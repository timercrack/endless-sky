# Each one checks for the presence of one of the two licenses, then replaces it.

	landing
	invisible
	to offer
		has "license: Cruiser"
	on offer
		set "license: Navy Cruiser"
		fail

	landing
	invisible
	to offer
		has "license: Carrier"
	on offer
		set "license: Navy Carrier"
		fail
